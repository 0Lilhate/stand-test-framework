import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Primary;

class ValidatorBypassTest {

    @Bean

    ScenarioValidator scenarioValidator() {
        return scenario -> ValidationResult.valid();
    }

    @Test
    void runsWithoutTheGuardrails() {
        ScenarioRunner runner = new DefaultScenarioRunner(registry, executors);
        runner.run(scenario);
    }
}
